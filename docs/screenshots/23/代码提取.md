## 1. `sort.glsl`：worklist_insert + main

```glsl
void worklist_insert(DdgiWorklistItem item) {
    index = subgroupBroadcastFirst(index) + subgroupExclusiveAdd(1);
    ddgi_worklist.lods[dispatch.lod].items[index] = item;
}

/// Updates a DDGI probe. Determines whether it should be active, and adds it to the appropriate worklist.
void main() {
    clear_object_buckets();

    ivec3 cell = ivec3(gl_GlobalInvocationID) + dispatch.base_position;
    DdgiProbeLocus probe = get_ddgi_probe(dispatch.lod, cell, object_buckets);

    u8vec3 offset = ddgi_probe_locus_offset(probe);
    DdgiProbeFlags flags = ddgi_probe_locus_flags(probe);

    set_probe_flags(ivec3(gl_LocalInvocationID), flags);
    load_adjacent_probes(cell);

    memoryBarrierShared();
    barrier();

    uint normalized_offset = ddgi_probe_pack_normalized_offset(fma(vec3(ddgi_inverse_scale(dispatch.lod)), vec3(offset) + vec3(0.5), vec3(-0.5)));
    ivec3 texel = probe_texel(cell);
    uint age = 0;

    if ((flags & DDGI_PROBE_FLAG_ENABLED) == DDGI_PROBE_FLAG_ENABLED && outside_lower_grid()) {
        vec3 cell_center = fma(vec3(dispatch.cell_length), vec3(cell), vec3(dispatch.cell_half_length));
        if (probe_near_surface(cell_center, flags)) {
            u16vec3 position = u16vec3(bitfieldInsert(offset, cell, int(DDGI_LOG2_PROBE_SPACING + dispatch.lod), int(32 - DDGI_LOG2_PROBE_SPACING - DDGI_LODS)));

            DdgiProbe previous = DdgiProbe(texelFetch(previous_ddgi_probes, texel, 0).x);
            bool reusable = all(lessThanEqual(dispatch.reuse_min_bound, cell))
                && all(lessThan(cell, dispatch.reuse_max_bound))
                && normalized_offset == ddgi_probe_packed_offset(previous);
            age = reusable ? ddgi_probe_age(previous) : 0;

            if (!can_skip_update(cell, cell_center, age)) {
                age = min(age + 1, 255);
                worklist_insert(ddgi_item_new(position, age));
            }
        }
    }

    imageStore(next_ddgi_probes, texel, uvec4(ddgi_probe_new(normalized_offset, age).offset_age, 0, 0, 0));
}
```

## 2. `sort.glsl`：main 里用到的两个判定函数（图 3）

```glsl
/// Gets the intersection of flags for all probes adjacent to `local_position`.
DdgiProbeFlags get_probe_flags_intersection() {
    ivec3 local_position = ivec3(gl_LocalInvocationID);
    ivec3 center = local_position + ivec3(1);
    return probe_flags[center.z][center.y][local_position.x]
        & probe_flags[center.z][center.y][center.x + 1]
        & probe_flags[center.z][local_position.y][center.x]
        & probe_flags[center.z][center.y + 1][center.x]
        & probe_flags[local_position.z][center.y][center.x]
        & probe_flags[center.z + 1][center.y][center.x];
}

/// Determines whether this invocation's probe is near a surface (either main grid or object).
/// If not, then the probe may be disabled.
bool probe_near_surface(vec3 cell_center, DdgiProbeFlags flags) {
    return (flags & DDGI_PROBE_FLAG_NO_SURFACES) == 0
        || (get_probe_flags_intersection() & DDGI_PROBE_FLAG_NO_SURFACES) == 0
        || probe_near_objects(cell_center);
}
```

## 3. collect 阶段：`collect_radiance`（图 4）

```glsl
/// Updates the irradiance value for a texel.
void collect_radiance(f16vec3 texel_direction, uint rays_per_probe) {
    f16vec4 result = f16vec4(0.0);
    for (uint i = 0; i < rays_per_probe; i++) {
        float16_t weight = max(float16_t(0.0), dot(texel_direction, directions[i]));
        result = fma(f16vec4(weight), f16vec4(radiances[i], 1.0), result);
    }

    f16vec3 irradiance = PI * result.xyz / result.w;
}
```

## 提取备注

- 图 1 中 `worklist_insert` 的第二行画面里以 `index =` 开头，**声明部分不可见**（左缘可能被裁，或该变量在外层作用域）；此处按画面原样保留。
- 图 2 的 `u16vec3 position = ...` 一行在编辑器里是软换行显示，本文按逻辑写成一行。
- 图 2 中 `if ((flags & DDGI_PROBE_FLAG_ENABLED) == ...)` 等行带底纹/删除线样式，是录制时的编辑器标注（搜索或 diff 高亮），不影响代码文本。
- `f16vec*` / `float16_t` 为 `GL_EXT_shader_explicit_arithmetic_types`（16 位浮点）写法。
