===== fn traverse_bit_set =====
fn traverse_bit_set(brick_entries: ptr<function, array<BrickMaskEntry, MAX_INTERSECTION_BUFFER_DEPTH>>, current_level: ptr<function, u32>,

    entry_index: ptr<function, u32>, encoded_position: ptr<function, EncodedPosition>, tag: ptr<function, u32>, options: MarchOptions, current_lod: u32, current_voxel: vec3<u32>) -> bool {

    *encoded_position = encode_position(current_voxel >> vec3<u32>(current_lod));

    *entry_index = decode_entry_index(*encoded_position, *current_level);

    while bit_set((*brick_entries)[*current_level].mask, *entry_index) {

        let compressed_index = pop_index((*brick_entries)[*current_level].mask, *entry_index);

        let entry_pointer = (*brick_entries)[*current_level].entry_pointer;

        let tag_pointer = entry_pointer + compressed_index + 2u;

        *tag = read_intersection_buffer(tag_pointer);



        if 0 < *current_level {

            *current_level -= 1u;

            (*brick_entries)[*current_level] = read_intersection_buffer_brick(entry_pointer + *tag);

            *entry_index = decode_entry_index(*encoded_position, *current_level);

        }

        else if march_through_enabled(options.march_through) {

            return march_through_tag_different(options.march_through, *tag);

        }

        else {

            return true;

        }

    }



    if march_through_enabled(options.march_through) {

        if all((*brick_entries)[*current_level].mask.values == vec2<u32>(0u)) {

            *tag = read_intersection_buffer((*brick_entries)[*current_level].entry_pointer + 2u);

            if march_through_tag_different(options.march_through, *tag) {

                return true;

            }

            else {

                *current_level += 1u;

                *entry_index = decode_entry_index(*encoded_position, *current_level);

                return false;

            }

            return true;

        }

        else {

            *tag = 0u;

            return true;

        }

    }

    else {

        return false;

    }

}

===== fn dda( =====
fn dda(entry_index: ptr<function, u32>, min_index: ptr<function, u32>, current_distance: ptr<function, f32>, current_voxel: ptr<function, vec3<u32>>,

    mask: BrickMask, direction: vec3<f32>, sides: vec3<f32>, step_size: f32) -> bool {

    var side_distance = sides;

    loop {

        let direction_component = direction[*min_index];

        side_distance[*min_index] += step_size / abs(direction_component);

        let changed_levels = increment_entry_voxel(entry_index, min_index, current_voxel, direction_component);        

        let has_voxel = bit_set(mask, *entry_index);

        

        if changed_levels || has_voxel {

            return changed_levels;

        }



        *min_index = min_component_vec3_index(side_distance);

        *current_distance = side_distance[*min_index];

    }



    return false;

}

===== fn march_intersection_buffer =====
fn march_intersection_buffer(command: MarchIntersectionBufferCommand, options: MarchOptions, current_distance: ptr<function, f32>, current_voxel: ptr<function, vec3<u32>>, result: ptr<function, MarchResult>) {

    var brick_entries: array<BrickMaskEntry, MAX_INTERSECTION_BUFFER_DEPTH>;

    brick_entries[MAX_INTERSECTION_BUFFER_LEVEL] = read_intersection_buffer_brick(command.intersection_pointer);

    var current_level = MAX_INTERSECTION_BUFFER_LEVEL;

    let direction_octant = encode_direction_octant(command.ray.direction);



    var entry_index: u32;

    var tag: u32;

    while (*current_distance < command.end_distance) && (current_level < MAX_INTERSECTION_BUFFER_DEPTH) {

        var encoded_position: EncodedPosition;

        if traverse_bit_set(&brick_entries, &current_level, &entry_index, &encoded_position, &tag, options, command.level_of_detail, *current_voxel) {

            break;

        }



        if !march_through_enabled(options.march_through) {

            let trace_mask = march_grid_data_buffer.march_masks[direction_octant.value][entry_index].xy;

            current_level += u32(all((brick_entries[current_level].mask.values & trace_mask) == vec2<u32>(0u)));

            entry_index = decode_entry_index(encoded_position, current_level);

        }



        let log2_step_length = (current_level << 1u) + command.level_of_detail;

        var side_distance = side_distance_for_ray(*current_voxel, log2_step_length, command.ray);

        var min_index = min_component_vec3_index(side_distance);



        if options.enable_max_unit_depth {

            let region_length = exp2(f32(log2_step_length));

            let max_level_depth = fma(options.max_unit_depth, region_length, -side_distance[min_index]);

            if max_level_depth < *current_distance {

                current_level = 0u;

                break;

            }

        }



        *current_distance = side_distance[min_index];



        if !march_through_enabled(options.march_through) && (current_level < MAX_INTERSECTION_BUFFER_DEPTH) {

            *current_voxel >>= vec3<u32>(log2_step_length);

            current_level += u32(dda(&entry_index, &min_index, current_distance, current_voxel, brick_entries[current_level].mask, command.ray.direction, side_distance, exp2(f32(log2_step_length))));

            *current_voxel <<= vec3<u32>(log2_step_length);

        }

        else {

            (*current_voxel)[min_index] += select(0xffffffffu, 1u, 0.0 <= command.ray.direction[min_index]) << log2_step_length;

        }



        if !march_through_enabled(options.march_through) && current_level == 0u {

            break;

        }

        

        let positive = 0.0 <= command.ray.direction[min_index];

        let mask = 0xffffffffu << ((current_level << 1u) + command.level_of_detail);

        let inverse_mask = ~mask;

        let component = ((*current_voxel)[min_index] & mask) | select(inverse_mask, 0u, positive);



        let new_level = (firstTrailingBit(component + select(1u, 0u, positive)) - command.level_of_detail) >> 1u;

        current_level = max(current_level, new_level);



        let base_voxel = *current_voxel & vec3<u32>(mask);

        *current_voxel = clamp(vec3<u32>(integer_floor(fma(vec3<f32>(*current_distance), command.ray.direction, command.ray.position))), base_voxel, base_voxel + vec3<u32>(inverse_mask));

        (*current_voxel)[min_index] = component;

    }



    if (*current_distance < command.end_distance) && (current_level < MAX_INTERSECTION_BUFFER_DEPTH) {

        if march_through_enabled(options.march_through) {

            (*result).tag = tag;

        }

        else if options.enable_max_unit_depth {

            (*result).tag = 1u;

        }

        else {

            let compressed_index = pop_index(brick_entries[0].mask, entry_index);

            let entry_pointer = brick_entries[0].entry_pointer;

            (*result).tag = read_intersection_buffer(entry_pointer + compressed_index + 2u);

        }



        (*result).distance = *current_distance;

        (*result).object_lod = command.object_lod;

        (*result).voxel = vec3<u32>(vec3<i32>(*current_voxel) >> vec3<u32>(command.level_of_detail));

    }

}

===== fn march( =====
fn march(command: MarchCommand, options: MarchOptions) -> MarchResult {

    var result: MarchResult;

    result.distance = command.end_distance;



    var current_distance = 0.0;

    var current_voxel = vec3<u32>(integer_floor(fma(vec3<f32>(current_distance), command.ray.direction, command.ray.position)));

    var current_lod = command.start_lod;



    while (current_distance < result.distance) && (current_lod < LEVELS_OF_DETAIL) {

        let grid_entry = read_grid(current_voxel, current_lod);



        if grid_entry.entry_pointer == EMPTY_INTERSECTION_POINTER {

            current_lod += 1u;

        }

        else {

            let initial_distance = current_distance;

            let object_lod = pack_object_lod(MAX_OBJECTS, current_lod);

            march_intersection_buffer(MarchIntersectionBufferCommand(result.distance, current_lod, grid_entry.entry_pointer, object_lod, command.ray),

                options, &current_distance, &current_voxel, &result);

            

            var object_slice = grid_entry.object_slice;

            while object_slice_not_empty(object_slice) {

                let object_command = march_command_for_object(read_increment_object_slice(&object_slice), command.ray, initial_distance, result.distance);

                if object_command.start_distance < object_command.end_distance {

                    march_object(object_command, options, &result);

                }

            }

        }

    }



    return result;

}

===== fn march_object =====
fn march_object(command: MarchObjectCommand, options: MarchOptions, result: ptr<function, MarchResult>) {

    var current_voxel = min(vec3<u32>(command.ray.position + command.start_distance * command.ray.direction), vec3<u32>(MAX_OBJECT_LENGTH - 1u));

    var current_distance = command.start_distance;

    let object_lod = pack_object_lod(command.object_id, 0u);

    march_intersection_buffer(MarchIntersectionBufferCommand(command.end_distance, 0u, command.intersection_pointer, object_lod, command.ray), options, &current_distance, &current_voxel, result);

}

===== fn march_command_for_object =====
fn march_command_for_object(object_id: u32, ray: Ray, start_distance: f32, end_distance: f32) -> MarchObjectCommand {

    let object_entry = &object_buffer[object_id];

    let transform = (*object_entry).transform;

    var command: MarchObjectCommand;



    command.ray.position = vec4<f32>(ray.position, 1.0) * transform;

    command.ray.direction = vec4<f32>(ray.direction, 0.0) * transform;



    let intersection_pointer_bounds = (*object_entry).intersection_pointer_bounds;

    command.intersection_pointer = intersection_pointer_bounds.x;

    command.object_id = object_id;



    let minimum_bound = extractBits(intersection_pointer_bounds.yzw, 0u, 8u);

    let maximum_bound = extractBits(intersection_pointer_bounds.yzw, 16u, 8u) + vec3<u32>(1u);

    

    command.start_distance = start_distance;

    command.end_distance = end_distance;



    project_marched_object(&command, minimum_bound, maximum_bound);



    return command;

}

===== fn read_intersection_buffer_brick =====

    return BrickMaskEntry(BrickMask(read_intersection_buffer_vec2(offset)), offset);

}

===== fn encode_direction_octant =====
fn encode_direction_octant(direction: vec3<f32>) -> Octant {

    let direction_cmp = vec3<u32>(vec3<f32>(0.0) <= direction);

    let direction_flags = direction_cmp << vec3<u32>(0u, 1u, 2u);

    return Octant(direction_flags.x | direction_flags.y | direction_flags.z);

}

===== fn side_distance_for_ray =====
fn side_distance_for_ray(current_voxel: vec3<u32>, log2_region_length: u32, ray: Ray) -> vec3<f32> {

    let update_increment_mask = vec2<u32>(1u, 0xffffffffu) << vec2<u32>(log2_region_length);

    let base = current_voxel & vec3<u32>(update_increment_mask.y);

    let bounds = select(vec3<f32>(0.0), vec3<f32>(f32(update_increment_mask.x)), vec3<f32>(0.0) <= ray.direction)

        + vec3<f32>(vec3<i32>(base)) - ray.position;

    return bounds / ray.direction;

}

===== fn step_log2_region_length =====
fn step_log2_region_length(current_level: u32, current_lod: u32, changed_levels: bool) -> u32 {

    return ((current_level + u32(changed_levels)) << 1u) + current_lod;

}

===== fn bit_set =====
fn bit_set(mask: BrickMask, index: u32) -> bool {

    let local_index = index & 31u;

    return bool((select(mask.values.y, mask.values.x, index < 32u) >> local_index) & 1u);

}

===== fn pop_index =====
fn pop_index(mask: BrickMask, index: u32) -> u32 {

    var local_full = select(mask.values.yx, vec2<u32>(mask.values.x, 0u), index < 32u);

    let local_index = index & 31u;

    local_full.x &= ~(0xffffffffu << local_index);

    let counted = countOneBits(local_full);

    return counted.x + counted.y;

}
